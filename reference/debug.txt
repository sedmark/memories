fetch('https://sedmark.github.io/memories/js/search_index.json')
  .then(r => {
    console.log('HTTP статус:', r.status);
    return r.text();
  })
  .then(txt => {
    console.log('Первые 500 символов файла:', txt.slice(0, 500));
    try {
      const data = JSON.parse(txt);
      console.log('Тип распарсенного:', Array.isArray(data) ? 'массив' : typeof data);
      console.log('Ключи верхнего уровня:', Array.isArray(data) ? '(массив)' : Object.keys(data));
    } catch (e) {
      console.error('Это не валидный JSON:', e.message);
    }

    const idx = lunr.Index.load(data);
    const results = idx.search('велик');
    console.log('Результаты поиска:', results);
  });
